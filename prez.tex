\section{Doktorat}

\begin{frame}
	\frametitle{Doktorat}
	\centering
	Cel projektu to opracowanie i wdrożenie metod ustalania pozycji bezzałogowego statku powietrznego bazujących na różnych źródłach danych, adaptujących się do warunków lotu, odpornych na zakłócenia. \\
	\vspace{20pt}
	Opracowane rozwiązanie ma stanowić alternatywę dla powszechnie wykorzystywanych systemów GNSS.
\end{frame}

\begin{frame}
	\frametitle{Platforma testowa -- WARMATE 3}
	\centering
	\vspace{10pt}
	\includegraphics[width=.6\textwidth]{warmate.jpg}
\end{frame}

\begin{frame}
	\frametitle{Wymagania stawiane proponowanym rozwiązaniom}
	\begin{itemize}
		\item Ograniczone wymiary i złożoność obliczeniowa
		\item Odporność na zakłócenia, niezależność
		\item Praca w różnych warunkach atmosferycznych
		\item Dokładność nawigacji porównywalna z systemem GNSS
		\item Minimalizacja negatywnego wpływu na sygnaturę cieplną i radiową platformy
	\end{itemize}
\end{frame}

\begin{frame}
	\frametitle{Zakres pracy}
		\begin{itemize}
		\item Przegląd dostępnej literatury naukowej i technicznej. Określenie luk w dostępnym stanie wiedzy.
		\item Badania w środowisku symulacyjnym oraz na platformie prototypowej
		\item Ocena skuteczności i dokładności wybranych metod
		\item Opracowanie algorytmów integracji danych uzyskanych z równoległego zastosowania wielu metod
		\item Publikacja wyników badań
	\end{itemize}
\end{frame}
